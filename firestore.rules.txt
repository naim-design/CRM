// ============================================================
// FIRESTORE SECURITY RULES
// Copy semua ni ke: Firebase Console -> Firestore Database -> Rules -> Publish
// ============================================================

rules_version = '2';
service cloud.firestore {
  // NOTA: Untuk koleksi "contacts" dengan puluhan ribu rekod, Firestore mungkin
  // minta awak buat "composite index" bila filter status+sumber sekali gus.
  // Kalau keluar error dalam browser console macam "The query requires an index",
  // ia akan bagi LINK terus — klik link tu, tekan "Create Index", tunggu ± 1 minit.
  match /databases/{database}/documents {

    // Semua user yang log masuk boleh baca & tulis.
    // (Untuk internal team tool, ni cukup selamat. Boleh diperketatkan lagi ikut role kalau perlu.)

    match /users/{userId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null && request.auth.uid == userId;
      allow update: if request.auth != null;
    }

    match /entries/{entryId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null;
      allow update, delete: if request.auth != null;
    }

    match /contacts/{contactId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null;
      allow update, delete: if request.auth != null;
    }

    match /todos/{todoId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null;
      allow update, delete: if request.auth != null;
    }

    match /posters/{posterId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null;
      allow update, delete: if request.auth != null;
    }

    match /meta/{docId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null;
    }

    match /importBatches/{batchId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null;
      allow update, delete: if request.auth != null;
    }

    match /feedback/{feedbackId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null;
      allow update, delete: if request.auth != null;
    }

    match /topups/{topupId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null;
      allow update, delete: if request.auth != null;
    }

    // Wabot webhook data is written by Firebase Admin (server-side).
    // Logged-in CRM users can read it, but browser clients cannot forge/write events.
    match /wabotEvents/{eventId} {
      allow read: if request.auth != null;
      allow write: if false;
    }

    match /wabotWebhookMeta/{docId} {
      allow read: if request.auth != null;
      allow write: if false;
    }
  }
}
